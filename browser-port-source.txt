//! Horizon: native Rust client for Minecraft Java Edition 26.3 gameplay.
#![forbid(unsafe_code)]
mod game;

use horizon_client::{crash, logging};

/// The mod API level this game serves, kept in the program for the launcher
/// to read without running it (`horizon_mods::api_level::MARK`). `main`
/// reads it, so the linker keeps it.
static MOD_API_LEVEL: [u8; 27] = horizon_mods::api_level::MARK;

fn main() {
    std::hint::black_box(&MOD_API_LEVEL);
    horizon_client::telemetry::mark_start();
    let args: Vec<String> = std::env::args().skip(1).collect();
    let game_args = args.iter().take_while(|a| *a != "--dedicated");
    let help = game_args.clone().any(|a| a == "--help" || a == "-h");
    // The dedicated server logs in its own folder, as vanilla's does; the game in its home.
    let (home, side, thread) = if args.iter().any(|a| a == "--dedicated") {
        (std::path::PathBuf::from("."), "server", "Server thread")
    } else {
        (logging::game_home(), "client", "Render thread")
    };
    if !help {
        logging::init(&home, thread);
        // The log's first line names the build, so a bug report that carries the log says which one it was.
        log::info!("Horizon {}", crash::version());
        crash::install(home.clone(), side);
    }
    if let Err(error) = game::run() {
        log::error!("Horizon: {error}");
        if !help {
            crash::report_error(&home, side, &*error);
        }
        // Started from Explorer, the console window closes with the game:
        // keep the message on screen until it is read.
        #[cfg(windows)]
        if std::io::IsTerminal::is_terminal(&std::io::stdin()) {
            eprintln!("Press Enter to close.");
            let _ = std::io::stdin().read_line(&mut String::new());
        }
        std::process::exit(1);
    }
    logging::flush();
}

//! The Microsoft account the launcher signed in, handed over in a session
//! file (docs/microsoft-account.md), and the Minecraft services call
//! online-mode servers need: authlib's MinecraftServicesSessionService.joinServer
//! through the services discovery document. The access token stays in this
//! process's memory: it is never logged, printed or written anywhere.
use std::{
    path::Path,
    sync::{Arc, Mutex, OnceLock},
    time::{Duration, SystemTime},
};

/// MinecraftServicesEnvironment.PROD.
pub const DEFAULT_DISCOVERY: &str = "https://discovery.minecraftservices.com/minecraft/client";
/// A session file older than this is refused: it can't be replayed later.
const MAX_SESSION_FILE_AGE: Duration = Duration::from_secs(120);

/// A signed-in account (vanilla's User with type msa).
pub struct AuthSession {
    pub name: String,
    pub uuid: u128,
    pub xuid: Option<String>,
    access_token: String,
}
impl std::fmt::Debug for AuthSession {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.debug_struct("AuthSession")
            .field("name", &self.name)
            .field("uuid", &horizon_protocol::crypto::uuid_string(self.uuid))
            .field("access_token", &"<hidden>")
            .finish()
    }
}

/// A UUID with or without dashes.
pub fn parse_uuid(text: &str) -> Option<u128> {
    let hex: String = text.chars().filter(|c| *c != '-').collect();
    (hex.len() == 32)
        .then(|| u128::from_str_radix(&hex, 16).ok())
        .flatten()
}

impl AuthSession {
    /// For tests and harnesses.
    pub fn new(name: &str, uuid: u128, access_token: &str) -> Self {
        Self {
            name: name.to_owned(),
            uuid,
            xuid: None,
            access_token: access_token.to_owned(),
        }
    }
    /// The session file's JSON: {"name", "uuid", "access_token", "user_type": "msa", "xuid"}.
    pub fn parse(text: &str) -> Result<Self, String> {
        let v: serde_json::Value =
            serde_json::from_str(text).map_err(|_| "the session file is not valid JSON")?;
        let field = |k: &str| v.get(k).and_then(|s| s.as_str());
        let name = field("name").ok_or("the session file has no name")?;
        if name.is_empty() || name.chars().count() > 16 {
            return Err("the session file's name is not a valid player name".into());
        }
        let uuid = field("uuid")
            .and_then(parse_uuid)
            .ok_or("the session file has no valid uuid")?;
        let access_token = field("access_token")
            .filter(|t| !t.is_empty())
            .ok_or("the session file has no access token")?;
        if field("user_type").is_some_and(|t| t != "msa") {
            return Err("the session file is not a Microsoft account session".into());
        }
        Ok(Self {
            name: name.to_owned(),
            uuid,
            xuid: field("xuid").map(str::to_owned),
            access_token: access_token.to_owned(),
        })
    }
    /// Read the session file the launcher wrote and delete it at once. It
    /// must be fresh and, on Unix, readable by its owner only.
    pub fn load(path: &Path) -> Result<Self, String> {
        let meta = std::fs::metadata(path).map_err(|e| format!("{}: {e}", path.display()))?;
        let refuse = |why: &str| {
            let _ = std::fs::remove_file(path);
            Err(format!("{}: {why}", path.display()))
        };
        #[cfg(unix)]
        {
            use std::os::unix::fs::PermissionsExt;
            if meta.permissions().mode() & 0o077 != 0 {
                return refuse("other users can read it");
            }
        }
        let age = meta
            .modified()
            .ok()
            .and_then(|m| SystemTime::now().duration_since(m).ok())
            .unwrap_or_default();
        if age > MAX_SESSION_FILE_AGE {
            return refuse("it is too old (start the game from the launcher)");
        }
        let text = std::fs::read_to_string(path);
        let _ = std::fs::remove_file(path);
        Self::parse(&text.map_err(|e| format!("{}: {e}", path.display()))?)
    }
}

static CURRENT: OnceLock<Arc<AuthSession>> = OnceLock::new();

/// The account this game was started with (set once at start).
pub fn set_current(session: AuthSession) {
    let _ = CURRENT.set(Arc::new(session));
}
pub fn current() -> Option<Arc<AuthSession>> {
    CURRENT.get().cloned()
}

/// Why joining a server through the session service failed, like the
/// AuthenticationException subclasses authlib throws.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum JoinError {
    /// No answer, or a 5xx (AuthenticationUnavailableException).
    Unavailable,
    /// 401 or ForbiddenOperationException (InvalidCredentialsException).
    InvalidSession,
    /// InsufficientPrivilegesException: multiplayer is off for the account.
    InsufficientPrivileges,
    /// multiplayer.access.banned or FORCED_USERNAME_CHANGE.
    Banned,
    /// Anything else, with the service's message.
    Other(String),
}
impl JoinError {
    /// MinecraftClientHttpException.toAuthenticationException.
    pub fn from_response(status: u16, body: &str) -> Self {
        let v: serde_json::Value = serde_json::from_str(body).unwrap_or_default();
        let error = v.get("error").and_then(|e| e.as_str()).unwrap_or("");
        match error {
            "ForbiddenOperationException" => return Self::InvalidSession,
            "multiplayer.access.banned" | "FORCED_USERNAME_CHANGE" => return Self::Banned,
            "InsufficientPrivilegesException" => return Self::InsufficientPrivileges,
            _ => {}
        }
        match status {
            401 => Self::InvalidSession,
            500.. => Self::Unavailable,
            _ => Self::Other(
                v.get("errorMessage")
                    .and_then(|m| m.as_str())
                    .map_or_else(|| format!("Status: {status}"), str::to_owned),
            ),
        }
    }
    /// ClientHandshakePacketListenerImpl.authenticateServer's message: the
    /// key of the reason inside "disconnect.loginFailedInfo", or its text.
    pub fn reason(&self) -> Result<&'static str, &str> {
        match self {
            Self::Unavailable => Ok("disconnect.loginFailedInfo.serversUnavailable"),
            Self::InvalidSession => Ok("disconnect.loginFailedInfo.invalidSession"),
            Self::InsufficientPrivileges => Ok("disconnect.loginFailedInfo.insufficientPrivileges"),
            Self::Banned => Ok("disconnect.loginFailedInfo.userBanned"),
            Self::Other(message) => Err(message),
        }
    }
}

/// The services discovery document URL: `HORIZON_API_DISCOVERY` overrides
/// it like authlib's `-Dminecraft.api.discovery.host`.
pub fn discovery_url() -> String {
    std::env::var("HORIZON_API_DISCOVERY")
        .ok()
        .filter(|u| !u.is_empty())
        .unwrap_or_else(|| DEFAULT_DISCOVERY.to_owned())
}

fn agent() -> ureq::Agent {
    ureq::AgentBuilder::new()
        .timeout(Duration::from_secs(15))
        .user_agent(&format!("Horizon/{}", horizon_core::release::VERSION))
        .build()
}

/// The discovery document, fetched once per game (a failure is retried next time).
pub(crate) fn discovery() -> Result<Arc<serde_json::Value>, JoinError> {
    static CACHE: Mutex<Option<Arc<serde_json::Value>>> = Mutex::new(None);
    if let Some(d) = CACHE.lock().unwrap().clone() {
        return Ok(d);
    }
    let doc: serde_json::Value = agent()
        .get(&discovery_url())
        .call()
        .ok()
        .and_then(|r| r.into_string().ok())
        .and_then(|t| serde_json::from_str(&t).ok())
        .ok_or(JoinError::Unavailable)?;
    let doc = Arc::new(doc);
    *CACHE.lock().unwrap() = Some(doc.clone());
    Ok(doc)
}

/// MinecraftServicesDiscoveryService.getUrl(service, endpoint).
pub fn endpoint(doc: &serde_json::Value, service: &str, name: &str) -> Option<String> {
    doc.pointer(&format!("/discovery/{service}/endpoints/{name}/uri"))
        .and_then(|u| u.as_str())
        .map(str::to_owned)
}

/// The JoinMinecraftServerRequest body; `selectedProfile` is the undashed UUID.
pub fn join_request(session: &AuthSession, server_hash: &str) -> String {
    serde_json::json!({
        "accessToken": session.access_token,
        "selectedProfile": format!("{:032x}", session.uuid),
        "serverId": server_hash,
    })
    .to_string()
}

/// MinecraftServicesSessionService.joinServer: tell the session service this
/// account is joining the server whose hash is `server_hash`.
pub fn join_server(session: &AuthSession, server_hash: &str) -> Result<(), JoinError> {
    let doc = discovery()?;
    let url = endpoint(&doc, "session", "join").ok_or(JoinError::Unavailable)?;
    match agent()
        .post(&url)
        .set("Content-Type", "application/json; charset=utf-8")
        .send_string(&join_request(session, server_hash))
    {
        Ok(_) => Ok(()),
        Err(ureq::Error::Status(status, response)) => Err(JoinError::from_response(
            status,
            &response.into_string().unwrap_or_default(),
        )),
        Err(ureq::Error::Transport(_)) => Err(JoinError::Unavailable),
    }
}

/// What the session service says about a player joining this server
/// (MinecraftServicesSessionService.hasJoinedServer).
#[derive(Clone, Debug, PartialEq)]
pub struct JoinedProfile {
    pub uuid: u128,
    pub properties: Vec<horizon_protocol::packets::login::Property>,
}

/// HttpDiscoveryService.buildQuery: `URLEncoder.encode(value, UTF-8)`.
fn url_encode(text: &str) -> String {
    let mut out = String::new();
    for b in text.bytes() {
        match b {
            b'a'..=b'z' | b'A'..=b'Z' | b'0'..=b'9' | b'.' | b'-' | b'*' | b'_' => {
                out.push(char::from(b));
            }
            b' ' => out.push('+'),
            _ => out.push_str(&format!("%{b:02X}")),
        }
    }
    out
}

/// The session service's verify endpoint, asked by an online-mode server
/// whether `name` joined with `server_hash` (and from `ip`, when the server
/// prevents proxy connections). `Ok(None)`: it didn't, or the service said
/// no; `Err(Unavailable)`: no answer.
pub fn has_joined(
    name: &str,
    server_hash: &str,
    ip: Option<std::net::IpAddr>,
) -> Result<Option<JoinedProfile>, JoinError> {
    let doc = discovery()?;
    let url = endpoint(&doc, "session", "verify").ok_or(JoinError::Unavailable)?;
    let mut query = format!(
        "username={}&serverId={}",
        url_encode(name),
        url_encode(server_hash)
    );
    if let Some(ip) = ip {
        query.push_str(&format!("&ip={}", url_encode(&ip.to_string())));
    }
    let separator = if url.contains('?') { '&' } else { '?' };
    let body = match agent().get(&format!("{url}{separator}{query}")).call() {
        Ok(response) => response.into_string().unwrap_or_default(),
        Err(ureq::Error::Status(status, response)) => {
            let error =
                JoinError::from_response(status, &response.into_string().unwrap_or_default());
            return match error {
                JoinError::Unavailable => Err(error),
                _ => Ok(None),
            };
        }
        Err(ureq::Error::Transport(_)) => return Err(JoinError::Unavailable),
    };
    Ok(parse_joined(&body))
}

/// HasJoinedMinecraftServerResponse: `id` and `properties`; no id is no profile.
pub fn parse_joined(body: &str) -> Option<JoinedProfile> {
    let v: serde_json::Value = serde_json::from_str(body).ok()?;
    let uuid = parse_uuid(v.get("id")?.as_str()?)?;
    let properties = v
        .get("properties")
        .and_then(|p| p.as_array())
        .into_iter()
        .flatten()
        .filter_map(|p| {
            Some(horizon_protocol::packets::login::Property {
                name: p.get("name")?.as_str()?.to_owned(),
                value: p.get("value")?.as_str()?.to_owned(),
                signature: p
                    .get("signature")
                    .and_then(|s| s.as_str())
                    .map(str::to_owned),
            })
        })
        .collect();
    Some(JoinedProfile { uuid, properties })
}

//! mods"). For a player who had one on, the game installs the mod from

/// Horizon's registry; `HORIZON_MOD_INDEX` points elsewhere (a local file
/// for trials, whose files may then be `file://`), as for the launcher.
pub const INDEX_URL: &str = "https://horizon-rust.github.io/horizon-mods/index.json";
const MAX_INDEX: u64 = 16 << 20;

/// An extra that is an official mod now.
#[derive(Debug, PartialEq)]
pub struct Moved {
    /// The mod's id in the registry.
    pub id: &'static str,
    /// The extra's name, for the line the player reads.
    pub name: &'static str,
}

pub const MOVED: [Moved; 4] = [
    Moved {
        id: "fps_overlay",
        name: "FPS counter",
    },
    Moved {
        id: "keystrokes",
        name: "Keystrokes",
    },
    Moved {
        id: "zoom",
        name: "Zoom",
    },
    Moved {
        id: "minimap",
        name: "Minimap",
    },
];

impl Moved {
    fn was_on(&self, e: &Extras) -> bool {
        match self.id {
            "fps_overlay" => e.fps,
            "keystrokes" => e.keystrokes,
            "zoom" => e.zoom,
            _ => e.minimap,
        }
    }
    /// The extra's settings as the mod keeps them (its storage keys).
    pub fn settings(&self, e: &Extras) -> Vec<(&'static str, String)> {
        let flag = |b: bool| if b { "true" } else { "false" }.to_owned();
        match self.id {
            "fps_overlay" => vec![
                ("corner", e.fps_corner.name().into()),
                ("lows", flag(e.fps_lows)),
            ],
            "keystrokes" => vec![
                ("corner", e.keystrokes_corner.name().into()),
                ("cps", flag(e.keystrokes_cps)),
            ],
            "zoom" => vec![
                ("level", e.zoom_level.to_string()),
                ("smooth", flag(e.zoom_smooth)),
                ("sensitivity", flag(e.zoom_sensitivity)),
            ],
            _ => vec![
                ("corner", e.minimap_corner.name().into()),
                ("size", e.minimap_size.to_string()),
                (
                    "shape",
                    if e.minimap_round { "round" } else { "square" }.into(),
                ),
                ("zoom", e.minimap_zoom.to_string()),
                ("rotate", flag(e.minimap_rotate)),
            ],
        }
    }
    /// The extra's key as the mod's key (`horizon-mods.txt`), when the
    /// player had changed it from the mod's default.
    pub fn keys(&self, e: &Extras) -> Vec<(&'static str, String)> {
        match self.id {
            "zoom" if e.zoom_key != "key.keyboard.z" => vec![("zoom", e.zoom_key.clone())],
            _ => Vec::new(),
        }
    }
    /// The line the player reads once it is installed.
    pub fn notice(&self) -> String {
        format!("{} moved to the mod browser: installed for you", self.name)
    }
}

/// The moved extras still to install: on, and not done yet.
pub fn pending(e: &Extras) -> Vec<&'static Moved> {
    MOVED
        .iter()
        .filter(|m| m.was_on(e) && !e.moved.iter().any(|d| d == m.id))
        .collect()
}

/// The moved extras that were off and not marked yet: nothing to install,
/// only to remember.
pub fn nothing_to_do(e: &Extras) -> Vec<&'static Moved> {
    MOVED
        .iter()
        .filter(|m| !m.was_on(e) && !e.moved.iter().any(|d| d == m.id))
        .collect()
}

fn location() -> String {
    std::env::var("HORIZON_MOD_INDEX")
        .ok()
        .filter(|v| !v.is_empty())
        .unwrap_or_else(|| INDEX_URL.to_owned())
}

/// `url`'s bytes, at most `max`: `https://` from the network; `file://` only
/// with a local index (a trial registry).
fn get(url: &str, local: bool, max: u64) -> Result<Vec<u8>, String> {
    let mut out = Vec::new();
    if url.starts_with("https://") {
        let response = ureq::AgentBuilder::new()
            .timeout(std::time::Duration::from_secs(60))
            .build()
            .get(url)
            .call()
            .map_err(|e| format!("{url}: {e}"))?;
        response
            .into_reader()
            .take(max + 1)
            .read_to_end(&mut out)
            .map_err(|e| e.to_string())?;
    } else {
        let path = match url.strip_prefix("file://") {
            Some(p) if local => p,
            _ if local && !url.contains("://") => url,
            _ => return Err(format!("{url}: not an https address")),
        };
        out = fs::read(path).map_err(|e| format!("{path}: {e}"))?;
    }
    if out.len() as u64 > max {
        return Err(format!("{url}: over {max} bytes"));
    }
    Ok(out)
}

/// The registry's index, from the network (or `HORIZON_MOD_INDEX`).
pub fn fetch_index() -> Result<Index, String> {
    let at = location();
    let local = !at.starts_with("https://");
    let text = String::from_utf8(get(&at, local, MAX_INDEX)?).map_err(|e| e.to_string())?;
    if local {
        Index::parse_local(&text)
    } else {
        Index::parse(&text)
    }
}

/// Installs `m` into the game folder `dir` (where `options.txt` is), for a
/// player whose extras were `e`: its newest version for this game that
/// wasn't taken down and that this Horizon's mod API level runs, checked
/// against the index (size, SHA-256, signature, manifest), as
/// `horizon-mods/<id>.hmod`, turned on, with the extra's settings in its
/// storage unless the mod has its own already.
pub fn install(index: &Index, m: &Moved, e: &Extras, dir: &Path) -> Result<(), String> {
    let project = index
        .project(m.id)
        .ok_or_else(|| format!("{} isn't in Horizon's registry yet", m.id))?;
    let version = index
        .installable(
            project,
            horizon_core::version::VERSION_ID,
            horizon_mods::api_level::LEVEL,
        )
        .ok_or_else(|| format!("{} has no version this Horizon can run", m.id))?;
    let mods = dir.join(crate::mod_list::DIR);
    fs::create_dir_all(&mods).map_err(|e| format!("{}: {e}", mods.display()))?;
    let target = mods.join(format!("{}.hmod", m.id));
    let current = index::sha256_file(&target).ok();
    if current.as_deref() != Some(version.file.sha256.as_str()) {
        let local = !location().starts_with("https://");
        let bytes = get(&version.file.url, local, index::MAX_FILE)?;
        let partial: PathBuf = mods.join(format!(".{}.hmod.download", m.id));
        fs::write(&partial, bytes).map_err(|e| format!("{}: {e}", partial.display()))?;
        if let Err(e) = index::check_file(index, project, version, &partial) {
            let _ = fs::remove_file(&partial);
            return Err(format!("{}: {e}", m.id));
        }
        fs::rename(&partial, &target).map_err(|e| format!("{}: {e}", target.display()))?;
    }
    let storage = mods.join(horizon_mods::manifest::STORAGE_DIR).join(m.id);
    fs::create_dir_all(&storage).map_err(|e| format!("{}: {e}", storage.display()))?;
    for (key, value) in m.settings(e) {
        let path = storage.join(key);
        if !path.exists() {
            fs::write(&path, value).map_err(|e| format!("{}: {e}", path.display()))?;
        }
    }
    let mut list = ModList::load(dir);
    list.enabled.insert(m.id.to_owned());
    // A key the player already chose for the mod stays.
    let keys = list.keys.entry(m.id.to_owned()).or_default();
    for (name, key) in m.keys(e) {
        keys.entry(name.to_owned()).or_insert(key);
    }
    list.keys.retain(|_, k| !k.is_empty());
    list.save(dir);
    Ok(())
}

//! Player skins from profile `textures` properties: authlib's
//! `MinecraftServicesSessionService.unpackTextures`, vanilla's
//! `SkinManager` lookup (which texture a player shows, and when the default
//! one) and `SkinTextureDownloader.processLegacySkin`.

/// `Property.hasSignature` and the services key check, as authlib's
/// `SignatureState`.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Signature {
    Unsigned,
    Signed,
    Invalid,
}

/// `MinecraftProfileTextures`: the texture URLs a profile names. Empty when
/// the payload doesn't decode or names a URL outside the allowed domains.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub struct Textures {
    pub skin: Option<String>,
    /// The skin's `metadata.model` is `slim` (`PlayerModelType.SLIM`).
    pub slim: bool,
    pub cape: Option<String>,
    pub elytra: Option<String>,
    pub signed: bool,
}

/// The `getTexture` endpoint's valid URIs from Minecraft services discovery.
pub const TEXTURE_URIS: [&str; 2] = [
    "https://textures.minecraft.net/texture/{textureId}",
    "http://textures.minecraft.net/texture/{textureId}",
];

/// `java.util.Base64.getDecoder().decode`.
pub fn base64_decode(text: &str) -> Option<Vec<u8>> {
    let mut out = Vec::with_capacity(text.len() / 4 * 3);
    let (mut bits, mut shift) = (0u32, 18i32);
    let bytes: Vec<char> = text.chars().collect();
    let mut i = 0;
    while i < bytes.len() {
        let c = bytes[i];
        i += 1;
        let b = match c {
            'A'..='Z' => c as u32 - 'A' as u32,
            'a'..='z' => c as u32 - 'a' as u32 + 26,
            '0'..='9' => c as u32 - '0' as u32 + 52,
            '+' => 62,
            '/' => 63,
            '=' => {
                if shift == 6 && (i == bytes.len() || bytes[i] != '=') || shift == 18 {
                    return None;
                }
                if shift == 6 {
                    i += 1;
                }
                break;
            }
            _ => return None,
        };
        bits |= b << shift;
        shift -= 6;
        if shift < 0 {
            out.extend([(bits >> 16) as u8, (bits >> 8) as u8, bits as u8]);
            shift = 18;
            bits = 0;
        }
    }
    match shift {
        6 => out.push((bits >> 16) as u8),
        0 => out.extend([(bits >> 16) as u8, (bits >> 8) as u8]),
        12 => return None,
        _ => {}
    }
    (i >= bytes.len()).then_some(out)
}

/// `unpackTextures(property)`: the textures of a `textures` property's
/// base64 JSON payload.
pub fn unpack(value: &str, signature: Signature, valid_uris: &[&str]) -> Textures {
    let empty = Textures::default();
    let Some(json) = base64_decode(value)
        .map(|b| String::from_utf8_lossy(&b).into_owned())
        .and_then(|s| serde_json::from_str::<serde_json::Value>(&s).ok())
    else {
        return empty;
    };
    let Some(textures) = json.get("textures").and_then(|t| t.as_object()) else {
        return empty;
    };
    if textures.is_empty() {
        return empty;
    }
    let mut out = Textures {
        signed: signature == Signature::Signed,
        ..empty.clone()
    };
    for (kind, texture) in textures {
        let Some(url) = texture.get("url").and_then(|u| u.as_str()) else {
            return empty;
        };
        let allowed = valid_uris
            .iter()
            .any(|u| url.starts_with(&u.replace("{textureId}", "")));
        if !allowed {
            return empty;
        }
        if kind == "SKIN" {
            out.slim = texture
                .get("metadata")
                .and_then(|m| m.get("model"))
                .and_then(|m| m.as_str())
                == Some("slim");
        }
        let slot = match kind.as_str() {
            "SKIN" => &mut out.skin,
            "CAPE" => &mut out.cape,
            "ELYTRA" => &mut out.elytra,
            _ => continue,
        };
        *slot = Some(url.to_owned());
    }
    out
}

/// The skin URL of a `minecraft:profile` component's property list. Player
/// heads and mannequins read their profile this way, and `SkinManager` lets
/// those textures be unsigned (`getInsecureSkinInformation`).
pub fn profile_skin_url(profile: &serde_json::Value) -> Option<String> {
    let (value, signature) = parse_profile(profile).textures?;
    unpack(&value, signature, &TEXTURE_URIS).skin
}
/// `profile_skin_url` on a block entity's `profile` NBT (`SkullBlockEntity`):
/// the same `properties` list, held as a compound. A name-only profile (a
/// string tag) has no properties and no skin URL.
pub fn profile_skin_url_nbt(profile: &horizon_world::nbt::Tag) -> Option<String> {
    let properties = match profile.get("properties")? {
        horizon_world::nbt::Tag::List(list) => list,
        _ => return None,
    };
    let value = properties.iter().find_map(|p| {
        let name = nbt_text(p.get("name"))?;
        (name == "textures")
            .then(|| nbt_text(p.get("value")))
            .flatten()
    })?;
    unpack(&value, Signature::Unsigned, &TEXTURE_URIS).skin
}

/// The profile's uuid as u128: `id` as an int array (block entity NBT) or a
/// uuid string (older and component forms). Used for `DefaultPlayerSkin.get`
/// when a head's skin is not downloaded yet.
pub fn profile_uuid_nbt(profile: &horizon_world::nbt::Tag) -> Option<u128> {
    match profile.get("id")? {
        horizon_world::nbt::Tag::IntArray(v) if v.len() == 4 => Some(
            v.iter()
                .fold(0u128, |a, &w| (a << 32) | u128::from(w as u32)),
        ),
        horizon_world::nbt::Tag::String(s) => {
            let digits: String = s.to_string_lossy().chars().filter(|c| *c != '-').collect();
            u128::from_str_radix(&digits, 16).ok()
        }
        _ => None,
    }
}

pub(crate) fn nbt_text(tag: Option<&horizon_world::nbt::Tag>) -> Option<String> {
    match tag {
        Some(horizon_world::nbt::Tag::String(s)) => Some(s.to_string_lossy()),
        _ => None,
    }
}

    Pending,
    /// The services' answer: `None` for no profile or no answer, which the
    /// cache holds like any result (`Optional.empty`).
    Done(Option<ResolvedProfile>),
}

/// `ProfileResolver.Cached`: resolves a `Dynamic` profile through the
/// services â€” `profileCacheByName` asks `profiles`/`getByName` for the id,
/// `profileCacheById` asks `session`/`getProfileById` for the profile â€”
/// at most 256 entries kept 10 minutes after their last use, fetched on
/// background threads.
pub struct Resolver {
    entries: std::collections::HashMap<NameOrId, Entry>,
    done: std::sync::mpsc::Receiver<(NameOrId, Option<ResolvedProfile>)>,
    send: std::sync::mpsc::Sender<(NameOrId, Option<ResolvedProfile>)>,
    last_sweep: std::time::Instant,
}
struct Entry {
    touched: std::time::Instant,
    /// `None` while the fetch is in flight.
    resolved: Option<Option<ResolvedProfile>>,
}
/// `ProfileResolver.Cached`'s cache bounds.
const MAX_PROFILES: usize = 256;
const PROFILE_TTL: std::time::Duration = std::time::Duration::from_secs(10 * 60);

/// `StringUtil.isValidPlayerName` (the check `fetchByName` makes first).
fn valid_player_name(name: &str) -> bool {
    name.encode_utf16().count() <= 16 && name.encode_utf16().all(|c| c > 32 && c < 127)
}

/// `fetchByNameOrId`: `getByName` answers `{id, name}`; `getProfileById`
/// (`fetchProfile(id, requireSecure)`) answers `{id, name, properties}` â€”
/// asked `unsigned=false` so the `textures` property is signed.
fn resolve_profile(key: &NameOrId, doc: &serde_json::Value) -> Option<ResolvedProfile> {
    let id = match key {
        NameOrId::Id(id) => *id,
        NameOrId::Name(name) => {
            let url = crate::auth::endpoint(doc, "profiles", "getByName")?
                .replace("{name}", &name.to_lowercase());
            let body = http_get(&url)?;
            let v: serde_json::Value = serde_json::from_slice(&body).ok()?;
            crate::auth::parse_uuid(v.get("id")?.as_str()?)?
        }
    };
    let url = crate::auth::endpoint(doc, "session", "getProfileById")?
        .replace("{profileId}", &format!("{id:032x}"));
    let separator = if url.contains('?') { '&' } else { '?' };
    let body = http_get(&format!("{url}{separator}unsigned=false"))?;
    let v: serde_json::Value = serde_json::from_slice(&body).ok()?;
    Some(ResolvedProfile {
        id: crate::auth::parse_uuid(v.get("id")?.as_str()?)?,
        name: v.get("name")?.as_str()?.to_owned(),
        properties: v
            .get("properties")?
            .as_array()?
            .iter()
            .map(|p| {
                Some((
                    p.get("name")?.as_str()?.to_owned(),
                    p.get("value")?.as_str()?.to_owned(),
                    p.get("signature")
                        .and_then(|s| s.as_str())
                        .map(str::to_owned),
                ))
            })
            .collect::<Option<Vec<_>>>()?,
    })
}

impl Default for Resolver {
    fn default() -> Self {
        Self::new()
    }
}
impl Resolver {
    pub fn new() -> Self {
        let (send, done) = std::sync::mpsc::channel();
        Self {
            entries: Default::default(),
            done,
            send,
            last_sweep: std::time::Instant::now(),
        }
    }
    /// `CacheBuilder.expireAfterAccess` with `maximumSize`.
    fn sweep(&mut self) {
        if self.last_sweep.elapsed() < std::time::Duration::from_secs(30) {
            return;
        }
        self.last_sweep = std::time::Instant::now();
        self.entries
            .retain(|_, e| e.touched.elapsed() < PROFILE_TTL);
        while self.entries.len() > MAX_PROFILES
            && let Some(oldest) = self
                .entries
                .iter()
                .min_by_key(|(_, e)| e.touched)
                .map(|(k, _)| k.clone())
        {
            self.entries.remove(&oldest);
        }
    }
    /// How the resolution of `key` stands, starting it the first time.
    pub fn state(&mut self, key: &NameOrId) -> Resolve {
        while let Ok((key, resolved)) = self.done.try_recv() {
            self.entries.insert(
                key,
                Entry {
                    touched: std::time::Instant::now(),
                    resolved: Some(resolved),
                },
            );
        }
        self.sweep();
        if let Some(entry) = self.entries.get_mut(key) {
            entry.touched = std::time::Instant::now();
            return match &entry.resolved {
                Some(resolved) => Resolve::Done(resolved.clone()),
                None => Resolve::Pending,
            };
        }
        // `fetchByName` refuses an invalid name without a fetch; like every
        // other empty answer that is the cached result.
        let resolved = matches!(key, NameOrId::Name(name) if !valid_player_name(name));
        self.entries.insert(
            key.clone(),
            Entry {
                touched: std::time::Instant::now(),
                resolved: resolved.then_some(None),
            },
        );
        if resolved {
            return Resolve::Done(None);
        }
        let (key, send) = (key.clone(), self.send.clone());
        std::thread::spawn(move || {
            let resolved = crate::auth::discovery()
                .ok()
                .and_then(|doc| resolve_profile(&key, &doc));
            let _ = send.send((key, resolved));
        });
        Resolve::Pending
    }
}

/// `ResolvableProfile` (an item's `minecraft:profile` component, a skull's
/// `profile` tag): the `textures` property it carries, the key a `Dynamic`
/// profile resolves by, and `partialProfile.id` â€” the fallback skin's key.
#[derive(Clone, Debug, Default, PartialEq)]
pub struct ProfileInfo {
    /// `GameProfile.id` (`DefaultPlayerSkin.get`).
    pub id: u128,
    /// The `textures` property's value and signature state, when present.
    pub textures: Option<(String, Signature)>,
    /// `Dynamic` profiles resolve by name or id.
    pub dynamic: Option<NameOrId>,
}

/// A profile's `id`: `UUIDUtil.CODEC` (an int array of four, or a UUID
/// string).
fn profile_id(v: &serde_json::Value) -> Option<u128> {
    match v {
        serde_json::Value::Array(a) => {
            let ints: Option<Vec<i64>> = a.iter().map(|x| x.as_i64()).collect();
            let i = ints?;
            (i.len() == 4).then(|| {
                ((i[0] as u32 as u128) << 96)
                    | ((i[1] as u32 as u128) << 64)
                    | ((i[2] as u32 as u128) << 32)
                    | i[3] as u32 as u128
            })
        }
        serde_json::Value::String(s) => crate::auth::parse_uuid(s),
        _ => None,
    }
}

/// `ResolvableProfile`'s read of a `profile` value: a bare name is
/// `createUnresolved` (the `PLAYER_NAME` codec's alternative), a compound
/// `name` + `id` + `properties` (`Partial.MAP_CODEC` or a `GameProfile`).
pub fn parse_profile(v: &serde_json::Value) -> ProfileInfo {
    if let serde_json::Value::String(name) = v {
        return ProfileInfo {
            id: horizon_protocol::crypto::offline_uuid(name),
            textures: None,
            dynamic: valid_player_name(name).then(|| NameOrId::Name(name.clone())),
        };
    }
    let name = v.get("name").and_then(|n| n.as_str()).map(str::to_owned);
    let id = v.get("id").and_then(profile_id);
    let properties = v.get("properties").and_then(|p| p.as_array());
    let textures = properties
        .into_iter()
        .flatten()
        .find(|p| p.get("name").and_then(|n| n.as_str()) == Some("textures"))
        .and_then(|p| {
            Some((
                p.get("value")?.as_str()?.to_owned(),
                if p.get("signature").is_some() {
                    Signature::Signed
                } else {
                    Signature::Unsigned
                },
            ))
        });
    // `ResolvableProfile.create`: a Partial is Dynamic with no properties
    // and exactly one of a name or an id.
    let dynamic =
        (properties.is_none_or(|p| p.is_empty()) && name.is_some() != id.is_some()).then(|| {
            name.as_deref().map_or_else(
                || NameOrId::Id(id.unwrap()),
                |n| NameOrId::Name(n.to_owned()),
            )
        });
    ProfileInfo {
        // `createPartialProfile`: the id, else the name's offline UUID, else nil.
        id: id
            .or_else(|| name.as_deref().map(horizon_protocol::crypto::offline_uuid))
            .unwrap_or(0),
        textures,
        dynamic,
    }
}

/// `ResolvableProfile.create`'s `Dynamic` key from a skull's `profile` tag:
/// a bare name string is `createUnresolved`, and a compound without
/// properties and with exactly one of a name or an id is `Dynamic`.
pub fn dynamic_nbt(profile: &horizon_world::nbt::Tag) -> Option<NameOrId> {
    if let horizon_world::nbt::Tag::String(s) = profile {
        let name = s.to_string_lossy();
        return valid_player_name(&name).then_some(NameOrId::Name(name));
    }
    if matches!(profile.get("properties"), Some(horizon_world::nbt::Tag::List(l)) if !l.is_empty())
    {
        return None;
    }
    match (nbt_text(profile.get("name")), profile_uuid_nbt(profile)) {
        (Some(name), None) => Some(NameOrId::Name(name)),
        (None, Some(id)) => Some(NameOrId::Id(id)),
        _ => None,
    }
}

/// `PlayerSkinRenderCache.getOrDefault` for a head: the resolved skin once
/// its texture is in the atlas (`atlas`), else `partialProfile`'s default.
pub fn head_skin(
    profile: &serde_json::Value,
    resolver: &mut Resolver,
    downloads: &mut Downloader,
    atlas: impl Fn(&str) -> bool,
) -> String {
    let p = parse_profile(profile);
    let property = p.textures.or_else(|| {
        let key = p.dynamic.as_ref()?;
        match resolver.state(key) {
            Resolve::Done(Some(profile)) => profile.textures(),
            _ => None,
        }
    });
    if let Some((value, signature)) = property
        && let Some(url) = unpack(&value, signature, &TEXTURE_URIS).skin
        && let Some(hash) = texture_hash(&url)
    {
        let location = skin_location(&hash);
        match downloads.state(Kind::Skin, &url) {
            Download::Ready if atlas(&location) => return location,
            // Its slot was evicted: offer the image again.
            Download::Ready => {
                downloads.again(&location);
            }
            _ => {}
        }
    }
    crate::tab_list::default_skin(p.id).to_owned()
}

/// `MinecraftProfileTexture.getHash`: the base name of the URL's path.
pub fn texture_hash(url: &str) -> Option<String> {
    let rest = url.split_once("://")?.1;
    let path = &rest[rest.find('/').unwrap_or(rest.len())..];
    let path = &path[..path.find(['?', '#']).unwrap_or(path.len())];
    let name = &path[path.rfind(['/', '\\']).map_or(0, |i| i + 1)..];
    Some(name[..name.rfind('.').unwrap_or(name.len())].to_owned())
}

/// `SkinManager.TextureCache.registerTexture`: the texture id of a skin,
/// `skins/` and the SHA-1 of its hash's UTF-16 code units.
pub fn skin_location(hash: &str) -> String {
    let units: Vec<u8> = hash.encode_utf16().flat_map(u16::to_le_bytes).collect();
    let digest = sha1(&units);
    let mut out = String::from("skins/");
    for b in digest {
        out.push_str(&format!("{b:02x}"));
    }
    out
}

/// Where a texture download stands.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Download {
    Ready,
    Pending,
    Failed,
}

/// `MinecraftProfileTexture.Type`: which texture cache a download is in.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub enum Kind {
    Skin,
    Cape,
    Elytra,
}

/// `SkinManager.createLookup(profile, requireSecure).get()`: the skin
/// texture a player shows, or `None` for their default skin. `download`
/// tells how the skin, cape and elytra texture downloads stand.
pub fn lookup(
    textures: &Textures,
    require_secure: bool,
    mut download: impl FnMut(Kind, &str) -> Download,
) -> Option<String> {
    let mut skin = None;
    let mut state = Download::Ready;
    let urls = [
        (Kind::Skin, &textures.skin),
        (Kind::Cape, &textures.cape),
        (Kind::Elytra, &textures.elytra),
    ];
    for (kind, url) in urls {
        let Some(url) = url else { continue };
        // registerTexture computes the hash first; a bad URL fails the lookup.
        let hash = texture_hash(url)?;
        if kind == Kind::Skin {
            skin = Some(skin_location(&hash));
        }
        match download(kind, url) {
            Download::Failed => return None,
            Download::Pending => state = Download::Pending,
            Download::Ready => {}
        }
    }
    if state != Download::Ready || require_secure && !textures.signed {
        return None;
    }
    skin
}

/// `SkinTextureDownloader.processLegacySkin` on RGBA pixels: 64x32 skins
/// get the 64x64 layout, and the opaque parts lose their alpha. `None` for
/// any other size (the download fails).
pub fn process_legacy_skin(width: u32, height: u32, rgba: &[u8]) -> Option<Vec<u8>> {
    if width != 64 || height != 32 && height != 64 {
        return None;
    }
    let legacy = height == 32;
    let mut img = vec![0u8; 64 * 64 * 4];
    let n = rgba.len().min(img.len());
    img[..n].copy_from_slice(&rgba[..n]);
    let at = |x: i32, y: i32| ((y * 64 + x) * 4) as usize;
    if legacy {
        let rects = [
            (4, 16, 16, 32, 4, 4),
            (8, 16, 16, 32, 4, 4),
            (0, 20, 24, 32, 4, 12),
            (4, 20, 16, 32, 4, 12),
            (8, 20, 8, 32, 4, 12),
            (12, 20, 16, 32, 4, 12),
            (44, 16, -8, 32, 4, 4),
            (48, 16, -8, 32, 4, 4),
            (40, 20, 0, 32, 4, 12),
            (44, 20, -8, 32, 4, 12),
            (48, 20, -16, 32, 4, 12),
            (52, 20, -8, 32, 4, 12),
        ];
        for (sx, sy, ox, oy, w, h) in rects {
            for y in 0..h {
                for x in 0..w {
                    let (s, d) = (at(sx + x, sy + y), at(sx + ox + w - 1 - x, sy + oy + y));
                    let p: [u8; 4] = img[s..s + 4].try_into().unwrap();
                    img[d..d + 4].copy_from_slice(&p);
                }
            }
        }
    }
    let no_alpha = |img: &mut Vec<u8>, x0: i32, y0: i32, x1: i32, y1: i32| {
        for x in x0..x1 {
            for y in y0..y1 {
                img[at(x, y) + 3] = 255;
            }
        }
    };
    no_alpha(&mut img, 0, 0, 32, 16);
    if legacy {
        // doNotchTransparencyHack: a fully opaque hat layer is cleared.
        let opaque = (32..64).all(|x| (0..32).all(|y| img[at(x, y) + 3] >= 128));
        if opaque {
            for x in 32..64 {
                for y in 0..32 {
                    img[at(x, y) + 3] = 0;
                }
            }
        }
    }
    no_alpha(&mut img, 0, 16, 64, 32);
    no_alpha(&mut img, 16, 48, 48, 64);
    Some(img)
}

/// `SkinTextureDownloader`: fetches textures on background threads, from
/// the local cache (`<root>/<2 hex>/<sha1>`) or over HTTP, keeping each one
/// by kind and hash as vanilla's `TextureCache`s do. A texture that can't
/// be fetched fails, and the player keeps their default skin.
pub struct Downloader {
    root: std::path::PathBuf,
    states: std::collections::HashMap<(Kind, String), Download>,
    done: std::sync::mpsc::Receiver<((Kind, String), Option<Vec<u8>>)>,
    send: std::sync::mpsc::Sender<((Kind, String), Option<Vec<u8>>)>,
    ready: Vec<(String, Vec<u8>)>,
    /// Locations already in `ready`, so `again` does not double-queue.
    queued: std::collections::HashSet<String>,
    /// Decoded skin images by location, kept so a skin evicted from the
    /// atlas uploads again (`again`); the disk cache backs anything dropped.
    images: std::collections::HashMap<String, Vec<u8>>,
    image_order: std::collections::VecDeque<String>,
}
/// `images` is bounded only so a long session on a busy server does not grow
/// without end; the disk cache holds everything it drops.
const MAX_SKIN_IMAGES: usize = 1024;
impl Downloader {
    pub fn new(root: std::path::PathBuf) -> Self {
        let (send, done) = std::sync::mpsc::channel();
        Self {
            root,
            states: Default::default(),
            done,
            send,
            ready: Vec::new(),
            queued: Default::default(),
            images: Default::default(),
            image_order: Default::default(),
        }
    }
    /// How the download of `url` stands, starting it the first time.
    pub fn state(&mut self, kind: Kind, url: &str) -> Download {
        while let Ok((key, image)) = self.done.try_recv() {
            let state = match image {
                Some(rgba) => {
                    if key.0 == Kind::Skin {
                        let location = skin_location(&key.1);
                        self.queue(location, rgba);
                    }
                    Download::Ready
                }
                None => Download::Failed,
            };
            self.states.insert(key, state);
        }
        let Some(hash) = texture_hash(url) else {
            return Download::Failed;
        };
        let key = (kind, hash);
        if let Some(state) = self.states.get(&key) {
            return *state;
        }
        self.states.insert(key.clone(), Download::Pending);
        let digest: String = skin_location(&key.1)["skins/".len()..].to_owned();
        let file = self
            .root
            .join(if digest.len() > 2 { &digest[..2] } else { "xx" })
            .join(&digest);
        let (url, send) = (url.to_owned(), self.send.clone());
        std::thread::spawn(move || {
            let bytes = std::fs::read(&file).ok().or_else(|| {
                let body = http_get(&url)?;
                let _ = std::fs::create_dir_all(file.parent()?);
                let _ = std::fs::write(&file, &body);
                Some(body)
            });
            let image = bytes
                .and_then(|b| crate::hud::decode_bytes(&b).ok())
                .and_then(|(w, h, rgba)| match kind {
                    Kind::Skin => process_legacy_skin(w, h, &rgba),
                    _ => Some(rgba),
                });
            let _ = send.send((key, image));
        });
        Download::Pending
    }
    /// Keeps a decoded skin for `again`, dropping the oldest past the bound.
    fn remember(&mut self, location: String, rgba: Vec<u8>) {
        if !self.images.contains_key(&location) {
            self.image_order.push_back(location.clone());
            while self.image_order.len() > MAX_SKIN_IMAGES
                && let Some(old) = self.image_order.pop_front()
            {
                self.images.remove(&old);
            }
        }
        self.images.insert(location, rgba);
    }
    /// An image off to the atlas: remembered for `again` and marked queued
    /// so nothing uploads the same location twice.
    fn queue(&mut self, location: String, rgba: Vec<u8>) {
        self.remember(location.clone(), rgba.clone());
        self.queued.insert(location.clone());
        self.ready.push((location, rgba));
    }
    /// A ready skin whose atlas slot was evicted (`show_skin` answered no):
    /// its image for upload again, from memory or the disk cache.
    pub fn again(&mut self, location: &str) -> bool {
        if self.queued.contains(location) {
            return true;
        }
        let Some(digest) = location.strip_prefix("skins/") else {
            return false;
        };
        let image = self.images.get(location).cloned().or_else(|| {
            let file = self
                .root
                .join(if digest.len() > 2 { &digest[..2] } else { "xx" })
                .join(digest);
            let (w, h, rgba) = crate::hud::decode_bytes(&std::fs::read(&file).ok()?).ok()?;
            process_legacy_skin(w, h, &rgba)
        });
        let Some(image) = image else {
            return false;
        };
        self.queue(location.to_owned(), image);
        true
    }
    /// Skins that finished since the last call: texture id and 64x64 RGBA.
    pub fn take_ready(&mut self) -> Vec<(String, Vec<u8>)> {
        for (location, _) in &self.ready {
            self.queued.remove(location);
        }
        std::mem::take(&mut self.ready)
    }
}

/// A GET over HTTP or HTTPS (textures.minecraft.net serves both; vanilla
/// uses https): the body of a 2xx answer, 16 MiB at most, after redirects.
fn http_get(url: &str) -> Option<Vec<u8>> {
    use std::io::Read;
    let response = ureq::AgentBuilder::new()
        .timeout(std::time::Duration::from_secs(15))
        .user_agent("Java")
        .build()
        .get(url)
        .call()
        .ok()?;
    let mut body = Vec::new();
    response
        .into_reader()
        .take(16 << 20)
        .read_to_end(&mut body)
        .ok()?;
    Some(body)
}

/// SHA-1 (FIPS 180-4).
pub fn sha1(data: &[u8]) -> [u8; 20] {
    let mut h: [u32; 5] = [
        0x6745_2301,
        0xEFCD_AB89,
        0x98BA_DCFE,
        0x1032_5476,
        0xC3D2_E1F0,
    ];
    let mut msg = data.to_vec();
    msg.push(0x80);
    while msg.len() % 64 != 56 {
        msg.push(0);
    }
    msg.extend((data.len() as u64 * 8).to_be_bytes());
    for block in msg.as_chunks::<64>().0 {
        let mut w = [0u32; 80];
        for (i, word) in block.as_chunks::<4>().0.iter().enumerate() {
            w[i] = u32::from_be_bytes(*word);
        }
        for i in 16..80 {
            w[i] = (w[i - 3] ^ w[i - 8] ^ w[i - 14] ^ w[i - 16]).rotate_left(1);
        }
        let [mut a, mut b, mut c, mut d, mut e] = h;
        for (i, wi) in w.iter().enumerate() {
            let (f, k) = match i {
                0..=19 => ((b & c) | (!b & d), 0x5A82_7999),
                20..=39 => (b ^ c ^ d, 0x6ED9_EBA1),
                40..=59 => ((b & c) | (b & d) | (c & d), 0x8F1B_BCDC),
                _ => (b ^ c ^ d, 0xCA62_C1D6),
            };
            let t = a
                .rotate_left(5)
                .wrapping_add(f)
                .wrapping_add(e)
                .wrapping_add(k)
                .wrapping_add(*wi);
            e = d;
            d = c;
            c = b.rotate_left(30);
            b = a;
            a = t;
        }
        for (x, y) in h.iter_mut().zip([a, b, c, d, e]) {
            *x = x.wrapping_add(y);
        }
    }
    let mut out = [0u8; 20];
    for (i, x) in h.iter().enumerate() {
        out[i * 4..i * 4 + 4].copy_from_slice(&x.to_be_bytes());
    }
    out
}

//! Server pack downloads into the game directory's `downloads/` cache, as
//! vanilla keeps it (`DownloadQueue`, `HttpUtil.downloadFile`,
//! `DownloadCacheCleaner`): `downloads/<pack id>/<SHA-1 of the file>`, one
//! batch at a time on a thread of its own, every download logged in
//! `downloads/log.json`, and the cache cut to its 20 newest files on start.
use super::{Batch, Request};
use sha1::{Digest, Sha1};
use std::{
    fs,
    io::{self, Read, Write},
    path::{Path, PathBuf},
    sync::mpsc::{Receiver, Sender, channel},
    time::SystemTime,
};

/// `DownloadedPackSource`'s MAX_PACK_SIZE_BYTES (250 MiB).
pub const MAX_SIZE: u64 = 262_144_000;
/// `DownloadQueue.MAX_KEPT_PACKS`.
const MAX_KEPT: usize = 20;

/// `createDownloadHeaders`: who is downloading, with which game.
pub fn headers(name: &str, uuid: u128) -> Vec<(&'static str, String)> {
    use horizon_core::version::{RESOURCE_PACK_FORMAT, VANILLA_VERSION, VERSION_ID};
    vec![
        ("X-Minecraft-Username", name.to_owned()),
        ("X-Minecraft-UUID", format!("{uuid:032x}")),
        ("X-Minecraft-Version", VANILLA_VERSION.to_owned()),
        ("X-Minecraft-Version-ID", VERSION_ID.to_owned()),
        (
            "X-Minecraft-Pack-Format",
            format!("{}.{}", RESOURCE_PACK_FORMAT.0, RESOURCE_PACK_FORMAT.1),
        ),
        ("User-Agent", format!("Minecraft Java/{VANILLA_VERSION}")),
    ]
}

/// `HttpUtil.DownloadProgressListener`, on the channel a batch was given.
#[derive(Clone, Copy, Debug, PartialEq)]
pub enum Progress {
    /// `requestStart`: the batch's next request begins.
    RequestStart,
    /// `downloadStart`: the response came, with its Content-Length if it
    /// said one.
    DownloadStart(Option<u64>),
    /// `downloadedBytes`: what this request has written so far.
    Bytes(u64),
    /// `requestFinished`: whether it succeeded.
    RequestFinished(bool),
}

type Job = (Vec<Request>, Vec<(&'static str, String)>, Sender<Progress>);

/// `DownloadQueue`: batches download in order, on one thread.
pub struct Queue {
    jobs: Sender<Job>,
    done: Receiver<Batch>,
}
impl Queue {
    pub fn open(dir: PathBuf) -> io::Result<Self> {
        fs::create_dir_all(&dir)?;
        vacuum(&dir, MAX_KEPT);
        let (jobs, todo) = channel::<Job>();
        let (finished, done) = channel();
        std::thread::Builder::new()
            .name("download-queue".into())
            .spawn(move || {
                for (requests, headers, progress) in todo {
                    if finished
                        .send(run(&dir, &requests, &headers, &progress))
                        .is_err()
                    {
                        break;
                    }
                }
            })?;
        Ok(Self { jobs, done })
    }
    pub fn download(
        &self,
        requests: Vec<Request>,
        headers: Vec<(&'static str, String)>,
        progress: Sender<Progress>,
    ) {
        let _ = self.jobs.send((requests, headers, progress));
    }
    /// The batches finished since the last call.
    pub fn finished(&self) -> Vec<Batch> {
        self.done.try_iter().collect()
    }
}

/// `DownloadQueue.runDownload`: each pack into `<dir>/<id>/`, logged.
fn run(
    dir: &Path,
    requests: &[Request],
    headers: &[(&'static str, String)],
    progress: &Sender<Progress>,
) -> Batch {
    let mut batch = Batch::default();
    for request in requests {
        let id = horizon_protocol::crypto::uuid_string(request.id);
        let _ = progress.send(Progress::RequestStart);
        let result = download_file(&dir.join(&id), request, headers, MAX_SIZE, progress);
        let _ = progress.send(Progress::RequestFinished(result.is_ok()));
        // DownloadQueue.LogEntry, its fields in the codec's order.
        let mut entry = vec![
            ("id", serde_json::Value::from(id)),
            ("url", request.url.as_str().into()),
            (
                "time",
                chrono::Utc::now()
                    .to_rfc3339_opts(chrono::SecondsFormat::AutoSi, true)
                    .into(),
            ),
        ];
        if let Some(sha1) = &request.sha1 {
            entry.push(("hash", sha1.as_str().into()));
        }
        entry.push(match &result {
            Ok(path) => match fs::metadata(path) {
                Ok(meta) => {
                    let name = path.strip_prefix(dir).unwrap_or(path);
                    let name = name.to_string_lossy().replace('\\', "/");
                    (
                        "file",
                        serde_json::json!({"name": name, "size": meta.len()}),
                    )
                }
                Err(_) => ("error", "no_access".into()),
            },
            Err(e) => {
                log::error!("Horizon: failed to download {}: {e}", request.url);
                ("error", "download_failed".into())
            }
        });
        if let Err(e) = append_log(&dir.join("log.json"), &entry) {
            log::error!("Horizon: failed to log download of {}: {e}", request.url);
        }
        match result {
            Ok(path) => batch.downloaded.push((request.id, path)),
            Err(_) => batch.failed.push(request.id),
        }
    }
    batch
}

/// `JsonEventLog.write`: one JSON object a line, as Gson writes it (with
/// its HTML escaping).
fn append_log(path: &Path, entry: &[(&str, serde_json::Value)]) -> io::Result<()> {
    let fields: Vec<String> = entry
        .iter()
        .map(|(k, v)| format!("{}:{v}", serde_json::Value::from(*k)))
        .collect();
    let mut line = String::new();
    for c in format!("{{{}}}", fields.join(",")).chars() {
        match c {
            '<' | '>' | '&' | '=' | '\'' => line.push_str(&format!("\\u{:04x}", c as u32)),
            c => line.push(c),
        }
    }
    line.push('\n');
    fs::OpenOptions::new()
        .create(true)
        .append(true)
        .open(path)?
        .write_all(line.as_bytes())
}

fn sha1_of(path: &Path) -> io::Result<String> {
    let mut hasher = Sha1::new();
    io::copy(&mut fs::File::open(path)?, &mut hasher)?;
    Ok(hex(&hasher.finalize()))
}
fn hex(bytes: &[u8]) -> String {
    bytes.iter().map(|b| format!("{b:02x}")).collect()
}
/// `checkExistingFile`: the file is there with that hash.
fn matches(path: &Path, sha1: &str) -> bool {
    path.exists() && sha1_of(path).is_ok_and(|actual| actual == sha1)
}
/// `updateModificationTime`: used now, so the cleaner keeps it longest.
fn touch(path: &Path) {
    let _ = fs::File::options()
        .append(true)
        .open(path)
        .and_then(|f| f.set_modified(SystemTime::now()));
}

/// `HttpUtil.downloadFile`: into `<dir>/<SHA-1>`. With a hash the cached file
/// is used when it matches, and a download that does not is an error;
/// without one the file is named after the hash of what came.
pub fn download_file(
    dir: &Path,
    request: &Request,
    headers: &[(&'static str, String)],
    max_size: u64,
    progress: &Sender<Progress>,
) -> Result<PathBuf, String> {
    let target = request.sha1.as_ref().map(|sha1| dir.join(sha1));
    if let (Some(target), Some(sha1)) = (&target, &request.sha1) {
        if matches(target, sha1) {
            log::info!("Horizon: returning cached file since actual hash matches requested");
            touch(target);
            return Ok(target.clone());
        }
        if target.exists() {
            log::warn!(
                "Horizon: existing file {} not found or had mismatched hash",
                target.display()
            );
        }
        fs::remove_file(target)
            .or_else(|e| {
                if e.kind() == io::ErrorKind::NotFound {
                    Ok(())
                } else {
                    Err(e)
                }
            })
            .map_err(|e| format!("Failed to remove existing file {}: {e}", target.display()))?;
    }
    let mut call = ureq::AgentBuilder::new().build().get(&request.url);
    for (name, value) in headers {
        call = call.set(name, value);
    }
    let response = call.call().map_err(|e| e.to_string())?;
    let size = response
        .header("Content-Length")
        .and_then(|v| v.trim().parse::<u64>().ok());
    fs::create_dir_all(dir).map_err(|e| e.to_string())?;
    let _ = progress.send(Progress::DownloadStart(size));
    if let Some(size) = size
        && size > max_size
    {
        return Err(format!(
            "Filesize is bigger than maximum allowed (file is {size}, limit is {max_size})"
        ));
    }
    let mut input = response.into_reader();
    let mut report = |written: u64| {
        let _ = progress.send(Progress::Bytes(written));
    };
    match (&target, &request.sha1) {
        (Some(target), Some(sha1)) => {
            let actual = save_hashed(&mut input, target, max_size, &mut report)
                .map_err(|e| e.to_string())?;
            if &actual != sha1 {
                return Err(format!(
                    "Hash of downloaded file ({actual}) did not match requested ({sha1})"
                ));
            }
            Ok(target.clone())
        }
        _ => {
            let tmp = dir.join(format!("download{:016x}.tmp", rand_suffix()));
            let result = save_hashed(&mut input, &tmp, max_size, &mut report).and_then(|actual| {
                let path = dir.join(&actual);
                if matches(&path, &actual) {
                    touch(&path);
                } else {
                    fs::rename(&tmp, &path)?;
                }
                Ok(path)
            });
            let _ = fs::remove_file(&tmp);
            result.map_err(|e| e.to_string())
        }
    }
}
fn rand_suffix() -> u64 {
    use std::hash::{BuildHasher, RandomState};
    RandomState::new().hash_one(std::time::Instant::now())
}
/// `downloadAndHash`: write what comes, at most `max_size` bytes.
fn save_hashed(
    input: &mut dyn Read,
    path: &Path,
    max_size: u64,
    progress: &mut dyn FnMut(u64),
) -> io::Result<String> {
    let mut out = fs::File::create(path)?;
    let mut hasher = Sha1::new();
    let mut buffer = [0u8; 8196];
    let mut read_so_far = 0u64;
    loop {
        let n = input.read(&mut buffer)?;
        if n == 0 {
            break;
        }
        read_so_far += n as u64;
        progress(read_so_far);
        if read_so_far > max_size {
            return Err(io::Error::other(format!(
                "Filesize was bigger than maximum allowed (got >= {read_so_far}, limit was {max_size})"
            )));
        }
        out.write_all(&buffer[..n])?;
        hasher.update(&buffer[..n]);
    }
    Ok(hex(&hasher.finalize()))
}

/// `DownloadCacheCleaner.vacuumCacheDir`: keep the `max_files` files most
/// worth keeping (in the pack folders: the newest, and one per pack before a
/// second of any), then drop the folders left empty.
pub fn vacuum(dir: &Path, max_files: usize) {
    let mut files = Vec::new();
    let mut stack = vec![dir.to_path_buf()];
    while let Some(d) = stack.pop() {
        let Ok(entries) = fs::read_dir(&d) else {
            continue;
        };
        for e in entries.flatten() {
            let path = e.path();
            let Ok(meta) = e.metadata() else { continue };
            if meta.is_dir() {
                stack.push(path);
            } else if meta.is_file() && path.parent() != Some(dir) {
                let time = meta.modified().unwrap_or(SystemTime::UNIX_EPOCH);
                files.push((path, time));
            }
        }
    }
    let Some(remove) = files.len().checked_sub(max_files).filter(|n| *n > 0) else {
        return;
    };
    // Newest first, then each file's rank in its folder (0 for a folder's
    // newest), reversed, and stably sorted by rank, highest first.
    files.sort_by_key(|f| std::cmp::Reverse(f.1));
    let mut counts = std::collections::HashMap::<PathBuf, usize>::new();
    let mut ranked: Vec<(PathBuf, usize)> = files
        .into_iter()
        .map(|(path, _)| {
            let count = counts
                .entry(path.parent().unwrap_or(dir).to_path_buf())
                .or_default();
            let rank = *count;
            *count += 1;
            (path, rank)
        })
        .collect();
    ranked.reverse();
    ranked.sort_by_key(|f| std::cmp::Reverse(f.1));
    let mut emptied = std::collections::HashSet::new();
    for (path, rank) in ranked.into_iter().take(remove) {
        match fs::remove_file(&path) {
            Ok(()) if rank == 0 => {
                emptied.insert(path.parent().unwrap_or(dir).to_path_buf());
            }
            Ok(()) => {}
            Err(e) => log::warn!(
                "Horizon: failed to delete cache file {}: {e}",
                path.display()
            ),
        }
    }
    emptied.remove(dir);
    for d in emptied {
        let _ = fs::remove_dir(d);
    }
}

    fn serve(status: &str, body: Vec<u8>) -> (String, std::thread::JoinHandle<String>) {
        let listener = TcpListener::bind("127.0.0.1:0").unwrap();
        let url = format!("http://{}/pack.zip", listener.local_addr().unwrap());
        let status = status.to_owned();
        let handle = std::thread::spawn(move || {
            let (mut stream, _) = listener.accept().unwrap();
            let mut request = Vec::new();
            let mut buf = [0u8; 1024];
            while !request.ends_with(b"\r\n\r\n") {
                let n = stream.read(&mut buf).unwrap();
                request.extend_from_slice(&buf[..n]);
            }
            let head = format!(
                "HTTP/1.1 {status}\r\nContent-Length: {}\r\nConnection: close\r\n\r\n",
                body.len()
            );
            stream.write_all(head.as_bytes()).unwrap();
            stream.write_all(&body).unwrap();
            String::from_utf8_lossy(&request).into_owned()
        });
        (url, handle)
    }

//! The anonymous "I'm playing" ping (`horizon_telemetry::live`, docs/telemetry.md) behind the
//! launcher's "N playing now". While the game runs, a thread of its own sends a random session
//! ID made here and kept only in memory, the version, the platform and whether a LAN or
//! play-with-friends world is open, every minute and at once when that last one changes.
//! It sends nothing when sending is turned off (`HORIZON_TELEMETRY_URL=off`) or with
//! `HORIZON_NO_HEARTBEAT=1` (automated runs).
use std::sync::OnceLock;
use std::sync::atomic::{AtomicBool, Ordering};
use std::thread::Thread;
use std::time::Duration;

use horizon_telemetry::live::{self, Ping};

/// A world is open to LAN or to friends.
static LAN: AtomicBool = AtomicBool::new(false);
static PINGER: OnceLock<Thread> = OnceLock::new();

/// After a failure the next ping waits longer, doubling up to this.
const MAX_WAIT: Duration = Duration::from_secs(600);

/// Starts the pings, once. Call when the game itself starts (not a dedicated server, not a
/// benchmark or an automated run).
pub fn start() {
    if PINGER.get().is_some() || std::env::var_os("HORIZON_NO_HEARTBEAT").is_some() {
        return;
    }
    let Some(endpoint) = horizon_telemetry::endpoint() else {
        return;
    };
    let spawned = std::thread::Builder::new()
        .name("Heartbeat".into())
        .spawn(move || run(&endpoint));
    match spawned {
        Ok(handle) => {
            let _ = PINGER.set(handle.thread().clone());
        }
        Err(e) => log::warn!("Horizon: no live count ping: {e}"),
    }
}

/// Whether a world is open to LAN or friends; a change is sent at once.
pub fn set_lan(open: bool) {
    if LAN.swap(open, Ordering::Relaxed) != open
        && let Some(pinger) = PINGER.get()
    {
        pinger.unpark();
    }
}

fn run(endpoint: &str) {
    let session = live::session_id();
    let version = crate::telemetry::version();
    let mut wait = live::INTERVAL;
    let mut failed = false;
    loop {
        let ping = Ping::new(&session, &version, LAN.load(Ordering::Relaxed));
        match live::send(endpoint, &ping) {
            Ok(()) => {
                wait = live::INTERVAL;
                failed = false;
            }
            Err(e) => {
                // Said once, not every minute the server is away.
                if !failed {
                    log::info!("Horizon: the live count ping failed ({e}); trying again later");
                }
                failed = true;
                wait = (wait * 2).min(MAX_WAIT);
            }
        }
        // Woken early by `set_lan`; a spurious wake only sends one ping more.
        std::thread::park_timeout(wait);
    }
}

/// StringUtil.isValidPlayerName: at most 16 UTF-16 units, none a control
/// character, a space or above ASCII.
pub fn valid_player_name(name: &str) -> bool {
    name.encode_utf16().count() <= 16 && name.encode_utf16().all(|c| c > 32 && c < 127)
}

/// How a server resolves a name it was given (CachedUserNameToIdResolver.get,
/// without the usercache.json cache): an offline server makes the offline
/// profile; an online one asks the services (for the name in lower case) and
/// knows nobody they don't.
pub fn resolve(name: &str, online: bool) -> Option<NameAndId> {
    if online && valid_player_name(name) {
        find_by_name(&discovery_url(), &name.to_ascii_lowercase())
    } else {
        (!online).then(|| NameAndId::offline(name))
    }
}

/// MinecraftServicesProfileRepository.findProfileByName: the account named
/// `name` (any case), or None when there is none or the services don't
/// answer (vanilla logs both as "Couldn't find profile with name").
pub fn find_by_name(discovery: &str, name: &str) -> Option<NameAndId> {
    let agent = ureq::AgentBuilder::new()
        .timeout(Duration::from_secs(15))
        .user_agent(&format!("Horizon/{}", horizon_core::release::VERSION))
        .build();
    let doc = json::parse(&agent.get(discovery).call().ok()?.into_string().ok()?, true).ok()?;
    let url = endpoint(&doc, "profiles", "getByName")?.replace("{name}", &name.to_lowercase());
    found(&agent.get(&url).call().ok()?.into_string().ok()?, name)
}

/// MinecraftServicesKeyInfo.get: the services' public keys from the
/// discovery document's `authentication` / `getPublicKeys` endpoint. The
/// profile-key keys (`playerCertificateKeys`, ServicesKeyType.PROFILE_KEY)
/// sign the players' chat keys; secure chat checks them. The error is the
/// line vanilla logs, after which it has no keys.
pub fn fetch_profile_key_keys(discovery: &str) -> Result<Vec<RsaPublicKey>, &'static str> {
    const FAILED: &str = "Failed to request Minecraft Services public key";
    let agent = ureq::AgentBuilder::new()
        .timeout(Duration::from_secs(15))
        .user_agent(&format!("Horizon/{}", horizon_core::release::VERSION))
        .build();
    let get = |url: &str| agent.get(url).call().ok()?.into_string().ok();
    let doc = get(discovery)
        .and_then(|d| json::parse(&d, true).ok())
        .ok_or(FAILED)?;
    let url = endpoint(&doc, "authentication", "getPublicKeys").ok_or(FAILED)?;
    profile_key_keys(&get(&url).ok_or(FAILED)?)
}

/// The key set answer: `{"profilePropertyKeys": [...], "playerCertificateKeys":
/// [...]}`, each `{"publicKey": base64 X.509}`. One bad key makes the whole
/// answer malformed (parseList throws).
pub fn profile_key_keys(answer: &str) -> Result<Vec<RsaPublicKey>, &'static str> {
    use base64::Engine;
    const MALFORMED: &str = "Received malformed Minecraft Services public key data";
    let doc = json::parse(answer, true).map_err(|_| MALFORMED)?;
    let list = |name: &str| -> Result<Vec<RsaPublicKey>, &'static str> {
        let Some(entries) = doc.get(name) else {
            return Ok(Vec::new());
        };
        let Json::Array(entries) = entries else {
            return Err(MALFORMED);
        };
        entries
            .iter()
            .map(|e| {
                let text = e
                    .get("publicKey")
                    .and_then(Json::as_string)
                    .ok_or(MALFORMED)?;
                let der = base64::engine::general_purpose::STANDARD
                    .decode(text)
                    .map_err(|_| MALFORMED)?;
                RsaPublicKey::from_der(&der).map_err(|_| MALFORMED)
            })
            .collect()
    };
    list("profilePropertyKeys")?;
    list("playerCertificateKeys")
}

/// MinecraftServicesDiscoveryService.getUrl(service, endpoint).
fn endpoint(doc: &Json, service: &str, name: &str) -> Option<String> {
    doc.get("discovery")?
        .get(service)?
        .get("endpoints")?
        .get(name)?
        .get("uri")?
        .as_string()
}

/// The answer, authlib's NameAndId: `{"id", "name"}`, the id undashed.
pub fn found(answer: &str, name: &str) -> Option<NameAndId> {
    let profile = json::parse(answer, true).ok()?;
    let found = profile.get("name")?.as_string()?;
    let id = profile.get("id")?.as_string()?.replace('-', "");
    (found.eq_ignore_ascii_case(name) && id.len() == 32).then_some(())?;
    Some(NameAndId {
        id: u128::from_str_radix(&id, 16).ok()?,
        name: found,
    })
}

/// `bytes` random bytes in lowercase hex.
pub(crate) fn random_hex(bytes: usize) -> String {
    let mut buf = vec![0u8; bytes];
    // Without the OS's randomness the time still makes names that don't collide in practice.
    if getrandom::fill(&mut buf).is_err() {
        let nanos = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .map_or(0, |d| d.as_nanos());
        for (i, b) in buf.iter_mut().enumerate() {
            *b = (nanos >> (8 * (i % 16))) as u8;
        }
    }
    buf.iter().map(|b| format!("{b:02x}")).collect()
}

/// Writes `bytes` through a temporary file, so a reader never sees half of it.
pub(crate) fn write_atomic(path: &Path, bytes: &[u8]) -> std::io::Result<()> {
    if let Some(parent) = path.parent() {
        std::fs::create_dir_all(parent)?;
    }
    let tmp = path.with_extension(format!("tmp-{}", random_hex(4)));
    std::fs::write(&tmp, bytes)?;
    std::fs::rename(&tmp, path)
}

//! The anonymous "I'm playing" ping, and the live count it feeds ("N playing now" in the
//! launcher).
//!
//! Unlike everything else in this crate it does not wait for consent: Ludovic decided
//! (2026-10-22) to count every player and tell them, in the launcher's first-run question, its
//! Settings and the privacy page. So a ping holds only what the count needs: a random session ID
//! made when the game starts and kept only in its memory (not the install ID), the Horizon
//! version, the platform and whether a LAN or play-with-friends world is open. The game sends
//! one every [`INTERVAL`]; the server counts a session while its last ping is under
//! [`WINDOW`] old, keeps nothing on disk and serves the counts rounded ([`round`]).
use std::collections::BTreeMap;
use std::time::Duration;

use serde::{Deserialize, Serialize};

/// How often the game pings while it runs.
pub const INTERVAL: Duration = Duration::from_secs(60);

/// How long a session counts after its last ping: two pings, so one lost on the way doesn't
/// drop it.
pub const WINDOW: Duration = Duration::from_secs(120);

/// The platforms a ping may name; anything else is `other`.
pub const PLATFORMS: [&str; 4] = ["linux", "macos", "windows", "other"];

/// The largest ping body the server reads.
pub const MAX_PING_BYTES: usize = 512;

const MAX_VERSION: usize = 40;

/// One "I'm playing", as it is sent: `{"session":"â€¦","version":"â€¦","platform":"â€¦","lan":false}`.
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
#[serde(deny_unknown_fields)]
pub struct Ping {
    /// 128 random bits in hex, made when the game starts ([`session_id`]) and never written down.
    pub session: String,
    /// The release (`v2026.10.15`, `0.2.0`) or `dev 0.1.0` for a build from source.
    pub version: String,
    /// One of [`PLATFORMS`].
    pub platform: String,
    /// A world is open to LAN or to friends over the internet.
    pub lan: bool,
}

impl Ping {
    pub fn new(session: &str, version: &str, lan: bool) -> Self {
        Self {
            session: session.to_owned(),
            version: version.to_owned(),
            platform: platform().to_owned(),
            lan,
        }
    }

    /// Why the server refuses this ping, if it does.
    pub fn check(&self) -> Result<(), &'static str> {
        if !crate::consent::is_install_id(&self.session) {
            return Err("session: 32 lowercase hex digits");
        }
        let version_ok = !self.version.is_empty()
            && self.version.len() <= MAX_VERSION
            && self
                .version
                .bytes()
                .all(|b| b.is_ascii_alphanumeric() || b" .+-_".contains(&b));
        if !version_ok {
            return Err("version: at most 40 letters, digits, spaces and .+-_");
        }
        if !PLATFORMS.contains(&self.platform.as_str()) {
            return Err("platform: linux, macos, windows or other");
        }
        Ok(())
    }
}

/// A new session ID: 128 random bits in hex.
pub fn session_id() -> String {
    crate::random_hex(16)
}

/// This machine's entry in [`PLATFORMS`].
pub fn platform() -> &'static str {
    match std::env::consts::OS {
        os @ ("linux" | "macos" | "windows") => os,
        _ => "other",
    }
}

/// The live count, as `GET /v1/live` serves it: every number already [`round`]ed.
#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
#[serde(default)]
pub struct Count {
    /// Sessions that pinged within the window.
    pub playing: u64,
    /// Of those, the ones with a LAN or play-with-friends world open.
    pub lan: u64,
    /// Per version: the most played ones, the rest summed under `other`.
    pub versions: BTreeMap<String, u64>,
    /// Per platform.
    pub platforms: BTreeMap<String, u64>,
}

/// Rounds a count for showing: exact under 10, then to the nearest 5 under 100, the nearest 10
/// under 1000 and the nearest 100 above. Small numbers stay true; large ones don't pretend to
/// a precision a two-minute window doesn't have.
pub fn round(n: u64) -> u64 {
    let step = match n {
        0..10 => return n,
        10..100 => 5,
        100..1000 => 10,
        _ => 100,
    };
    (n + step / 2) / step * step
}

/// Sends `ping` to `endpoint` (`https://host`, without a trailing slash).
#[cfg(feature = "live")]
pub fn send(endpoint: &str, ping: &Ping) -> Result<(), String> {
    agent()
        .post(&format!("{endpoint}/v1/live"))
        .set("Content-Type", "application/json")
        .send_string(&serde_json::to_string(ping).map_err(|e| e.to_string())?)
        .map(drop)
        .map_err(|e| e.to_string())
}

/// The live count at `endpoint`.
#[cfg(feature = "live")]
pub fn fetch(endpoint: &str) -> Result<Count, String> {
    let body = agent()
        .get(&format!("{endpoint}/v1/live"))
        .call()
        .map_err(|e| e.to_string())?
        .into_string()
        .map_err(|e| e.to_string())?;
    serde_json::from_str(&body).map_err(|e| e.to_string())
}

#[cfg(feature = "live")]
fn agent() -> ureq::Agent {
    ureq::AgentBuilder::new()
        .timeout(Duration::from_secs(5))
        .build()
}

#[cfg(test)]
mod tests {
    use super::*;

    fn ping() -> Ping {
        Ping::new(&session_id(), "v2026.10.15", false)
    }

    #[test]
    fn a_ping_holds_only_the_four_fields() {
        let json = serde_json::to_value(Ping::new(&"a".repeat(32), "0.2.0", true)).unwrap();
        let mut keys: Vec<&str> = json
            .as_object()
            .unwrap()
            .keys()
            .map(|k| k.as_str())
            .collect();
        keys.sort_unstable();
        assert_eq!(keys, ["lan", "platform", "session", "version"]);
        assert!(serde_json::to_string(&ping()).unwrap().len() < MAX_PING_BYTES);
    }

    #[test]
    fn session_ids_are_fresh_each_time() {
        let (a, b) = (session_id(), session_id());
        assert_ne!(a, b);
        assert!(ping().check().is_ok());
    }

    #[test]
    fn bad_pings_are_refused() {
        let bad = |f: fn(&mut Ping)| {
            let mut p = ping();
            f(&mut p);
            p.check().is_err()
        };
        assert!(bad(|p| p.session = "ABC".into()));
        assert!(bad(|p| p.session = "G".repeat(32)));
        assert!(bad(|p| p.version = String::new()));
        assert!(bad(|p| p.version = "1".repeat(41)));
        assert!(bad(|p| p.version = "<script>".into()));
        assert!(bad(|p| p.platform = "haiku".into()));
        assert!(!bad(|p| p.version = "dev 0.1.0".into()));
        let extra = r#"{"session":"00000000000000000000000000000000","version":"0.2.0","platform":"linux","lan":false,"name":"Steve"}"#;
        assert!(serde_json::from_str::<Ping>(extra).is_err());
    }

    #[test]
    fn counts_round_more_as_they_grow() {
        let cases = [
            (0, 0),
            (1, 1),
            (9, 9),
            (10, 10),
            (12, 10),
            (13, 15),
            (97, 95),
            (98, 100),
            (104, 100),
            (105, 110),
            (994, 990),
            (995, 1000),
            (1049, 1000),
            (1050, 1100),
        ];
        for (n, shown) in cases {
            assert_eq!(round(n), shown, "{n}");
        }
    }
}
